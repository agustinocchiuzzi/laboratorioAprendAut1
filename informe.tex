% TODO — 130-150 palabras de texto (además de las 2 ecuaciones,
% que ya están puestas abajo y no hace falta reescribir)
% - Formalización: entropía, ganancia de información.
% - Rol del hiperparámetro min\_info\_gain como criterio de corte.
% - Manejo de casos borde (subconjunto vacío, valores no vistos en test).
\begin{equation}
H(S) = -\sum_{c \in \{L,V,E\}} p_c \log_2 p_c
\label{eq:entropia}
\end{equation}
\begin{equation}
IG(S, A) = H(S) - \sum_{v \in valores(A)} \frac{|S_v|}{|S|} H(S_v)
\label{eq:ganancia}
\end{equation}
